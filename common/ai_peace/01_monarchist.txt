ai_monarchism = {
	enable = {
		ideology = monarchism
	}
	annex_randomness = 0
	liberate_randomness = 0
	puppet_randomness = 0
	take_states_randomness = 0
	force_government_randomness = 0
	
	# ROOT = Taker, FROM = Giver
	annex = {
		factor = 5
    modifier = {
      factor = 0
			FROM = {
				NOT = {
					OR = { 
						is_claimed_by = ROOT
						is_core_of = ROOT
					}
				}
			}
    }
    modifier = {
      factor = 999
      FROM = {
        OR = {
          is_claimed_by = ROOT
          is_core_of = ROOT
        }
      }
    }
	}

	# ROOT = Taker, FROM = Liberated
	liberate = { # Only liberate friends
		factor = 0 #100
	}

	# ROOT = Taker, FROM = Giver
	puppet = { # Monarchies almost always puppet
		factor = 200
		modifier = { #Don't puppet nations in civil war... this is too buggy.
			factor = 0
			FROM = { has_civil_war = yes }
		}
	}
	
	# ROOT = Taker, FROM = Giver
	puppet_all = {
		base = 0 
	}
	
	# ROOT = STATE, FROM = Taker, FROM.FROM = Giver
	puppet_state = {
		base = 0
	}

	# ROOT = Taker, FROM = State
	take_states = { # Monarchs prefer vassals...
		factor = 0

		modifier = { # ... unless the land is rightfully theirs
			factor = 100
			FROM = {
				OR = {
					is_claimed_by = ROOT
					is_core_of = ROOT
				}
			}
		}
		modifier = { #Only take claimed/core land if a puppet.
			factor = 0
			is_puppet = yes
			FROM = {
				NOT = {
					OR = { 
						is_claimed_by = ROOT
						is_core_of = ROOT
					}
				}
			}
		}
	}
	force_government = { # Why have a free Monarchy when you could do some nation building?
		factor = 0
	}
}
